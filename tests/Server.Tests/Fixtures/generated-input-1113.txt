################################################################################
# PDD SCORER — PTW Roos 34001 geometry, 2 mm depth bins
#
# Scores the central-axis PDD over the first 5 cm of the cylindrical water tank.
#
# RoosPDD:
#   - radius = 0.78 cm
#   - length = 5.0 cm
#   - ZBins = 25
#   - depth-bin thickness = 5.0/25 = 0.2 cm = 2 mm
#
# Each scoring bin therefore represents a cylindrical sensitive volume with
# the 7.8 mm Roos chamber radius and 2 mm thickness.
#
# WaterTank:
#   - centre at world z = +15.0 cm
#   - front surface at world z = 0.0 cm
#
# RoosPDD:
#   - local TransZ = -12.5 cm
#   - world centre = +2.5 cm
#   - scoring range = 0 to 5 cm depth
################################################################################

s:Ge/RoosPDD/Type         = "TsCylinder"
s:Ge/RoosPDD/Parent       = "WaterTank"
b:Ge/RoosPDD/IsParallel   = "True"

s:Ge/RoosPDD/Color        = "red"
s:Ge/RoosPDD/DrawingStyle = "Wireframe"

d:Ge/RoosPDD/RMin         = 0.0 cm
d:Ge/RoosPDD/RMax         = 0.78 cm
d:Ge/RoosPDD/HL           = 2.5 cm

d:Ge/RoosPDD/SPhi         = 0.0 deg
d:Ge/RoosPDD/DPhi         = 360.0 deg

d:Ge/RoosPDD/TransX       = 0.0 cm
d:Ge/RoosPDD/TransY       = 0.0 cm
d:Ge/RoosPDD/TransZ       = -12.5 cm

s:Sc/PDD/Quantity                  = "DoseToMedium"
s:Sc/PDD/Component                 = "RoosPDD"

i:Sc/PDD/RBins                     = 1
i:Sc/PDD/PhiBins                   = 1
i:Sc/PDD/ZBins                     = 25

sv:Sc/PDD/Report                   = 1 "Sum"
s:Sc/PDD/OutputType                = "CSV"
s:Sc/PDD/OutputFile                = "/srv/cluster/tsebt/runs/1113/seed11137_phsp21_pdd"
s:Sc/PDD/IfOutputFileAlreadyExists = "Overwrite"
